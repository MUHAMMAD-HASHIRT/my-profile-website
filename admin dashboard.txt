        // Initialize on page load
        function initializePage() {
            // Check if user is logged in
            if (!sessionStorage.getItem('adminLoggedIn')) {
                window.location.href = 'admin-login.html';
                return;
            }

            // Set user name
            const userEmail = sessionStorage.getItem('userEmail');
            if (userEmail) {
                const userNameEl = document.getElementById('userName');
                if (userNameEl) {
                    userNameEl.textContent = userEmail;
                }
            }

            // Wait a bit for data-storage.js to load, then update stats
            let attempts = 0;
            const checkDataStorage = setInterval(function() {
                attempts++;
                
                if (typeof schoolData !== 'undefined') {
                    console.log('Data storage found! Updating stats...');
                    clearInterval(checkDataStorage);
                    
                    try {
                        const data = schoolData.getData();
                        console.log('Initial data check - Students:', data.students ? data.students.length : 0);
                        
                        // Update stats
                        updateStats();
                    } catch (e) {
                        console.error('Error checking initial data:', e);
                    }
                } else if (attempts > 10) {
                    console.error('data-storage.js not loaded after 10 attempts!');
                    clearInterval(checkDataStorage);
                    document.getElementById('totalStudentsCount').textContent = 'Error';
                    document.getElementById('studentsChange').textContent = 'Data not loaded';
                }
            }, 200);
        }

        // Wait for DOM to be ready
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', initializePage);
        } else {
            // DOM already loaded
            initializePage();
        }
        
        // Also run on window load as backup
        window.addEventListener('load', function() {
            setTimeout(() => {
                if (typeof schoolData !== 'undefined') {
                    console.log('Window loaded - Updating stats again');
                    updateStats();
                }
            }, 300);
        });

        function showAddStudentModal() {
            console.log('showAddStudentModal called');
            const modal = document.getElementById('addStudentModal');
            if (!modal) {
                alert('Error: Modal not found. Please refresh the page.');
                console.error('addStudentModal element not found');
                return;
            }
            
            console.log('Showing modal');
            modal.style.display = 'flex';
            
            // Clear all fields
            const idField = document.getElementById('newStudentId');
            const nameField = document.getElementById('newStudentName');
            const classField = document.getElementById('newStudentClass');
            const sectionField = document.getElementById('newStudentSection');
            
            if (idField) idField.value = '';
            if (nameField) nameField.value = '';
            if (classField) classField.value = '';
            if (sectionField) sectionField.value = '';
            
            // Focus on first field
            setTimeout(() => {
                if (idField) idField.focus();

        function showSection(section) {

            const dashboardSection = document.getElementById('dashboardSection');

        function loadStudentsList() {
            console.log('loadStudentsList called');
            
            // Check if data-storage is loaded
            if (typeof schoolData === 'undefined') {
                console.error('data-storage.js not loaded!');
                const tbody = document.getElementById('studentsTableBody');
                if (tbody) {
                    tbody.innerHTML = 
                        '<tr><td colspan="5" style="text-align: center; padding: 20px; color: #c33;">Error: Data storage not loaded. Please refresh the page.</td></tr>';
                }
                return;
            }
            
            // Get the table body element
            const tbody = document.getElementById('studentsTableBody');
            if (!tbody) {
                console.error('studentsTableBody element not found!');
                // Try to find it again after a delay
                setTimeout(() => {
                    const retryTbody = document.getElementById('studentsTableBody');
                    if (retryTbody) {
                        console.log('Found tbody on retry, loading students...');
                        loadStudentsList();
                    } else {
                        console.error('Still cannot find studentsTableBody');
                    }
                }, 500);
                return;
            }
            
            try {
                // Get data from storage
                const data = schoolData.getData();
                console.log('Data retrieved:', data);
                
                if (!data) {
                    console.error('No data returned from getData()');
                    tbody.innerHTML = '<tr><td colspan="5" style="text-align: center; padding: 20px; color: #c33;">Error: Could not load data.</td></tr>';
                    return;
                }
                
                // Ensure students array exists
                if (!data.students) {
                    console.log('students array not found, creating it...');
                    data.students = [];
                    schoolData.saveData(data);
                }
                
                console.log('Students count:', data.students.length);
                console.log('Students data:', JSON.stringify(data.students));
                
                // Update student count display
                const countElement = document.getElementById('studentCount');
                if (countElement) {
                    countElement.textContent = data.students.length;
                }
                
                // If no students, show message
                if (data.students.length === 0) {
                    tbody.innerHTML = '<tr><td colspan="7" style="text-align: center; padding: 20px; color: #666;">No students found. Go to "Student Details" section to add one.</td></tr>';
                    return;
                }
                
                // Build table rows
                let html = '';
                data.students.forEach((student, index) => {
                    console.log(`Processing student ${index}:`, student);
                    html += `<tr>
                        <td><strong style="color: #1a237e; font-family: monospace;">${student.studentId || 'N/A'}</strong></td>
                        <td>${student.name || 'N/A'}</td>
                        <td>Class ${student.class || 'N/A'}</td>
                        <td>Section ${student.section || 'N/A'}</td>
                        <td>${student.phone || '-'}</td>
                        <td style="max-width: 200px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${student.address || ''}">${student.address || '-'}</td>

                            <button class="btn btn-primary" style="padding: 5px 10px; font-size: 12px;" onclick="editStudent('${student.id}')">Edit</button>

                console.log('Table updated with', data.students.length, 'students');
                
            } catch (error) {
                console.error('Error loading students:', error);
                console.error('Error stack:', error.stack);
                if (tbody) {
                    tbody.innerHTML = 
                        '<tr><td colspan="5" style="text-align: center; padding: 20px; color: #c33;">Error loading students: ' + error.message + '</td></tr>';
                }
            }

        function loadTeachersList() {
            const data = schoolData.getData();
            const container = document.getElementById('teachersList');
            
            if (data.teachers.length === 0) {
                container.innerHTML = '<p style="text-align: center; padding: 20px; color: #666;">No teachers found.</p>';
                return;
            }
            
            let html = '<table class="table" style="margin-top: 20px;">
                <thead>
                    <tr>
                        <th>Teacher Name</th>
                        <th>Subject</th>
                        <th>Portal ID</th>
                        <th>Password</th>
                        <th>Email</th>
                        <th>Assigned Classes</th>
                        <th>Actions</th>
                    </tr>
                </thead>
                <tbody>';
            
            data.teachers.forEach(teacher => {
                const classes = teacher.classes || [];
                const portalId = teacher.portalId || 'Not Assigned';
                const password = teacher.password || 'Not Set';
                
                html += `<tr>
                    <td><strong style="color: #1a237e;">${teacher.name}</strong></td>
                    <td><span style="background: #e3f2fd; color: #1a237e; padding: 4px 12px; border-radius: 15px; font-size: 12px; font-weight: 600;">${teacher.subject}</span></td>
                    <td><code style="background: #f5f5f5; padding: 4px 8px; border-radius: 4px; font-family: monospace;">${portalId}</code></td>
                    <td>
                        <span id="pwd_${teacher.id}" style="font-family: monospace;">${'*'.repeat(password.length)}</span>
                        <button class="btn" onclick="togglePassword('${teacher.id}')" style="padding: 2px 8px; font-size: 11px; margin-left: 5px; background: #6c757d; color: white;">Show</button>
                    </td>
                    <td style="font-size: 13px; color: #666;">${teacher.email}</td>
                    <td>`;
                
                if (classes.length === 0) {
                    html += '<span style="color: #999; font-style: italic;">None</span>';
                } else {
                    classes.forEach((cls, idx) => {
                        if (idx > 0) html += ', ';
                        html += `<span style="background: #1a237e; color: white; padding: 2px 8px; border-radius: 12px; font-size: 11px; margin-right: 3px;">${cls.class}${cls.section}</span>`;
                    });
                }
                
                html += `</td>
                    <td>
                        <button class="btn btn-primary" onclick="editTeacherCredentials('${teacher.id}')" style="padding: 5px 10px; font-size: 12px;">Edit ID/Pass</button>
                        <button class="btn btn-primary" onclick="window.location.href='admin-management.html'" style="padding: 5px 10px; font-size: 12px; margin-left: 5px;">Assign Classes</button>
                    </td>
                </tr>`;
            });
            
            html += '</tbody></table>';
            container.innerHTML = html;
        }

        function togglePassword(teacherId) {
            const data = schoolData.getData();
            const teacher = data.teachers.find(t => t.id === teacherId);
            if (!teacher) return;
            
            const pwdElement = document.getElementById('pwd_' + teacherId);
            const btn = event.target;
            
            if (btn.textContent === 'Show') {
                pwdElement.textContent = teacher.password || 'Not Set';
                btn.textContent = 'Hide';
            } else {
                const pwd = teacher.password || '';
                pwdElement.textContent = '*'.repeat(pwd.length || 8);
                btn.textContent = 'Show';
            }
        }

        function editTeacherCredentials(teacherId) {
            const data = schoolData.getData();
            const teacher = data.teachers.find(t => t.id === teacherId);
            
            if (!teacher) return;
            
            const newPortalId = prompt('Enter new Portal ID:', teacher.portalId || '');
            if (newPortalId === null) return; // User cancelled
            
            const newPassword = prompt('Enter new Password:', teacher.password || '');
            if (newPassword === null) return; // User cancelled
            
            if (!newPortalId.trim() || !newPassword.trim()) {
                alert('Portal ID and Password cannot be empty!');
                return;
            }
            
            // Check if Portal ID already exists
            const idExists = data.teachers.some(t => t.id !== teacherId && t.portalId === newPortalId.trim());
            if (idExists) {
                alert('Portal ID already exists. Please use a different ID.');
                return;
            }
            
            teacher.portalId = newPortalId.trim();
            teacher.password = newPassword.trim();
            
            schoolData.saveData(data);
            alert('Teacher credentials updated successfully!');
            loadTeachersList();
        }

        function updateStudentFromDetails(studentId) {
            if (typeof schoolData === 'undefined') {
                alert('Error: Data storage not loaded.');
                return;
            }

            const studentIdValue = document.getElementById('detailStudentId').value.trim();
            const name = document.getElementById('detailStudentName').value.trim();
            const classNum = document.getElementById('detailStudentClass').value;
            const section = document.getElementById('detailStudentSection').value;
            const phone = document.getElementById('detailStudentPhone').value.trim();
            const address = document.getElementById('detailStudentAddress').value.trim();

            if (!studentIdValue || !name || !classNum || !section) {
                alert('Please fill all required fields');
                return;
            }

            try {
                const data = schoolData.getData();
                const student = data.students.find(s => s.id === studentId);
                
                if (student) {
                    // Check if new ID conflicts with another student
                    const idConflict = data.students.some(s => s.id !== studentId && s.studentId && s.studentId.toLowerCase() === studentIdValue.toLowerCase());
                    if (idConflict) {
                        alert('Student ID already exists. Please use a different ID.');
                        return;
                    }
                    
                    student.studentId = studentIdValue;
                    student.name = name;
                    student.class = classNum;
                    student.section = section;
                    student.phone = phone || '';
                    student.address = address || '';
                    
                    schoolData.saveData(data);
                    alert('Student updated successfully!');
                    
                    // Reset form
                    clearStudentForm();
                    
                    // Reset button
                    const saveBtn = document.querySelector('#studentDetailsForm button.btn-primary');
                    if (saveBtn) {
                        saveBtn.textContent = 'Save Student';
                        saveBtn.onclick = saveStudentDetails;
                    }
                    
                    // Update stats and reload list
                    updateStats();
                    if (document.getElementById('studentsSection').style.display !== 'none') {
                        loadStudentsList();
                    }
                } else {
                    alert('Student not found!');
                }
            } catch (error) {
                console.error('Error updating student:', error);
                alert('Error updating student: ' + error.message);
            }
        }

        function updateStats() {
            console.log('updateStats called');
            
            if (typeof schoolData === 'undefined') {
                console.error('schoolData not available for stats update');
                const totalStudentsEl = document.getElementById('totalStudentsCount');
                const studentsChangeEl = document.getElementById('studentsChange');
                if (totalStudentsEl) totalStudentsEl.textContent = '0';
                if (studentsChangeEl) studentsChangeEl.textContent = 'Data not loaded';
                return;
            }
            
            try {
                const data = schoolData.getData();
                console.log('Data retrieved for stats:', data);
                
                if (!data) {
                    console.error('No data returned');
                    return;
                }
                
                const studentCount = (data.students && data.students.length) || 0;
                const teacherCount = (data.teachers && data.teachers.length) || 0;
                
                console.log('Student count:', studentCount, 'Teacher count:', teacherCount);
                
                // Update the stat cards
                const totalStudentsEl = document.getElementById('totalStudentsCount');
                if (totalStudentsEl) {
                    totalStudentsEl.textContent = studentCount.toLocaleString();
                    console.log('Updated totalStudentsCount to:', studentCount);
                } else {
                    console.error('totalStudentsCount element not found');
                }
                
                const studentsChangeEl = document.getElementById('studentsChange');
                if (studentsChangeEl) {
                    if (studentCount > 0) {
                        studentsChangeEl.textContent = `${studentCount} registered student${studentCount !== 1 ? 's' : ''}`;
                    } else {
                        studentsChangeEl.textContent = 'No students registered yet';
                    }
                    console.log('Updated studentsChange text');
                } else {
                    console.error('studentsChange element not found');
                }
                
                // Update other stat cards if they exist
                const statCards = document.querySelectorAll('.stat-value');
                if (statCards.length > 0) {
                    // First card is students (if totalStudentsCount doesn't exist)
                    if (!totalStudentsEl && statCards[0]) {
                        statCards[0].textContent = studentCount.toLocaleString();
                    }
                    // Second card is teachers
                    if (statCards.length > 1 && statCards[1]) {
                        statCards[1].textContent = teacherCount;
                    }
                }
                
                console.log('Stats updated successfully - Students:', studentCount, 'Teachers:', teacherCount);
            } catch (error) {
                console.error('Error updating stats:', error);
                console.error('Error stack:', error.stack);
                
                // Show error in UI
                const totalStudentsEl = document.getElementById('totalStudentsCount');
                const studentsChangeEl = document.getElementById('studentsChange');
                if (totalStudentsEl) totalStudentsEl.textContent = 'Error';
                if (studentsChangeEl) studentsChangeEl.textContent = 'Error loading data';
            }
        }
